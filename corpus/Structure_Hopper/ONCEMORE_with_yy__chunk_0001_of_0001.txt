# --- "BEATER" (PRODUCTION-READY) ---
use_bpm 120
set_volume! 1.0
use_debug false
set :song_status, :running # Global flag for graceful stops

# --- HARMONY DEFINITIONS (V4.5.1 COMPATIBLE) ---
define :chorus_chords do
  # Pop Progression: I-V-vi-IV (C-G-Am-F)
  (ring chord(:C3, :major), chord(:G2, :major), chord(:A2, :m7), chord(:F2, :M7))
end

define :verse_chords do
  # Altered Pop: vi-IV-I-V (Am-F-C-G)
  (ring chord(:A2, :m7), chord(:F2, :M7), chord(:C3, :major), chord(:G2, :major))
end

# --- EXPERIMENTAL BRIDGE: TIMBRE CHORDS ---
define :play_timbre_chord do |note, mods|
  mods.each do |m|
    synth :fm, note: note, mod_index: m, divisor: 2, amp: 0.4, release: 2
  end
end

# --- CORE INSTRUMENTATION LOOPS ---
live_loop :beat_anchor do
  stop unless get(:song_status) == :running
  # Kick drum evolves amplitude over 16 beats
  k_amp = (line 0.7, 1.2, steps: 16).mirror.tick
  sample :bd_haus, amp: k_amp
  
  # Glitch percussion
  sample :elec_plip, amp: 0.4 if one_in(3)
  sample :elec_twang, rate: 1.5, amp: 0.3 if one_in(4)
  
  sleep 0.5
end

live_loop :beat_depth do
  stop unless get(:song_status) == :running
  sleep 5
  # Bar counter to emphasize first beat
  bar_pos = tick(:bar_counter) % 4
  
  # Kick: emphasize first beat, subtle randomization
  k_amp = 0.7 + (bar_pos == 0 ? 0.2 : 0) + rrand(-0.01, 0.01) if one_in(10)
  sample :bd_haus, amp: k_amp
  
  # Glitch percussion with micro timing drift
  sleep 0.50 + rrand(-0.02, 0.01) # slight swing/drift
  sample :elec_plip, amp: 0.4 + rrand(-0.05,0.05) if one_in(6)
  
  sleep 0.49 + rrand(-0.01, 0.01)
  sample :elec_twang, rate: 1.5, amp: 0.3 + rrand(-0.05,0.05) if one_in(2)
  
  sleep 0.51 + rrand(-0.02, 0.01)
  # Optional ghost hit
  sample :elec_tick, amp: 0.2 + rrand(-0.05,0.01) if one_in(4)
  
end

# --- HARMONY LAYER (SMOOTH & LUSH) ---
live_loop :harmony_layer do
  stop unless get(:song_status) == :running
  
  current_chord = (look < 32 ? verse_chords.tick : chorus_chords.tick)
  
  # Layered: staggered blade notes
  use_synth :blade
  current_chord.each do |n|
    play n, release: rrand(2.0, 2.5), amp: 0.25, cutoff: 70, pan: rrand(-0.2, 0.2)
    sleep 0.05
  end
  
  # Sub-layer for warmth
  use_synth :hollow
  current_chord.each do |n|
    play n - 12, release: 3.0, amp: 0.15, cutoff: 60
  end
  
  sleep 2
end


live_loop :glitch_melody do
  stop unless get(:song_status) == :running
  use_synth :pluck
  
  # Define a motif
  motif = ring(:E4, :G4, :A4, :C5, :r)
  cycle_count = tick(:cycle_counter) + motif.length  # count how many times motif has run
  
  motif.length.times do
    n = motif.tick(:motif_tick)
    
    # Skip rests
    next if n == :r
    
    # Dynamic amplitude: evolves over cycles
    amp = case cycle_count
    when 0
      rrand(1.0, 1.1)  # almost mechanical
    when 1
      rrand(1.5, 0.8)  # subtle emphasis, slightly more expressive
    else
      rrand(1.6, 1.0)  # fully in the groove, more expressive
    end
    
    # Timing: evolve from tight to swing
    base_sleep = rrand(1.0, 1.01)
    sleep_offset = case cycle_count
    when 0
      rrand(2.95, 2.71)          # tiny hesitation
    when 1
      (tick(:swing_tick) % 4 == 0 ? 0.01 : -0.01)  # micro swing
    else
      (tick(:swing_tick) % 2 == 0 ? 0.02 : -0.02)  # intra-beat microswing
    end
    
    play n, amp: amp, release: rrand(1.15, 2.25), pan: rrand(-0.1, 0.1)
    sleep base_sleep + sleep_offset
  end
end




# --- SONG CONTROL THREAD ---
in_thread do
  sleep 64
  
  # --- EXPERIMENTAL BRIDGE ---
  puts "--- BRIDGE: TIMBRE TRANSITION ---"
  with_bpm 90 do  # less drastic than 60
    4.times do
      play_timbre_chord :A3, [0.8, 1.2, 1.5]
      sleep 2
    end
    
    # DISSONANT SHOCK: C# Major 7
    puts "--- DISSONANT PEAK ---"
    chord(:Cs4, :M7).each do |n|
      synth :bass_foundation, note: n, release: 4, amp: 0.25
      sleep 0.05
    end
    # Subtle sub-layer
    chord(:Cs4, :M7).each do |n|
      synth :mod_tri, note: n - 12, release: 4, amp: 0.15
    end
    sleep 4
  end
  
  # RESOLUTION
  puts "--- RESOLUTION ---"
  chord(:A2, :madd9).each do |n|
    synth :piano, note: n, release: 8, amp: 0.25
    sleep 0.05
  end
  # Sub-layer for warmth
  chord(:A2, :madd9).each do |n|
    synth :hollow, note: n - 12, release: 8, amp: 0.15
  end
  sleep 8
  
  
  # STOP EVERYTHING GRACEFULLY
  set :song_status, :finished
end
