# BLACKLIGHT RAINBOW

#!/usr/bin/env python3
#"""
#Blacklight Rainbow (thin-arc version)

#Process:
#--------
#1. Load an input image, resize to 512×512.
#2. Choose one vertical column of pixels (random by default).
#3. Draw 512 concentric circles, starting at an outer radius of 3950 px, decreasing by exactly 1 px per step.
#4. Render on an 8100×4000 canvas, with the #circle center placed at the bottom edge. This produces a narrow rainbow arc near the top of the canvas.

#Parameters:
#-----------
#- canvas_w:   8100 (default)
#- canvas_h:   4000 (default)
#- outer_radius: 3950 (default)
#- line_width: 6 (stroke width in px)
#- seed: random seed for reproducible slice choice
#- slice_x: optional fixed column index (0–511). If None, chosen randomly.
"""

import argparse
import random
from PIL import Image, ImageDraw

def render_rainbow(
    input_path: str,
    canvas_w: int = 8100,
    canvas_h: int = 4000,
    outer_radius: int = 3950,
    line_width: int = 6,
    seed: int | None = None,
    slice_x: int | None = None,
):
    # Seed RNG for reproducibility
    if seed is not None:
        random.seed(seed)

    # Load and normalize input image
    ref_img = Image.open(input_path).convert("RGB").resize((512, 512))

    # Pick vertical slice
    if slice_x is None:
        slice_x = random.randint(0, 511)
    elif not (0 <= slice_x <= 511):
        raise ValueError("slice_x must be in [0, 511]")

    # Prepare canvas
    canvas = Image.new("RGB", (canvas_w, canvas_h), "black")
    draw = ImageDraw.Draw(canvas, "RGB")

    cx = canvas_w // 2
    cy = canvas_h  # center at bottom edge

    # Draw concentric circles
    for i in range(512):
        r = outer_radius - i
        color = ref_img.getpixel((slice_x, i))

        bbox = [int(cx - r), int(cy - r), int(cx + r), int(cy + r)]
        draw.ellipse(bbox, outline=color, width=line_width)

    return canvas, slice_x

def main():
    parser = argparse.ArgumentParser(description="Blacklight Rainbow renderer (thin arc version).")
    parser.add_argument("--input", type=str, required=True, help="Path to input image.")
    parser.add_argument("--output", type=str, default="blacklight_rainbow.png", help="Path to save output image.")
    parser.add_argument("--canvas_w", type=int, default=8100)
    parser.add_argument("--canvas_h", type=int, default=4000)
    parser.add_argument("--outer_radius", type=int, default=3950)
    parser.add_argument("--line_width", type=int, default=6)
    parser.add_argument("--seed", type=int, default=None)
    parser.add_argument("--slice_x", type=int, default=None)
    args = parser.parse_args()

    img, used_slice = render_rainbow(
        input_path=args.input,
        canvas_w=args.canvas_w,
        canvas_h=args.canvas_h,
        outer_radius=args.outer_radius,
        line_width=args.line_width,
        seed=args.seed,
        slice_x=args.slice_x,
    )
    img.save(args.output)
    print(f"Saved: {args.output}")
    print(f"Slice column used: {used_slice}")

if __name__ == "__main__":
    main()
